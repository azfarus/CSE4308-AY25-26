    stock_quantity INT

    order_id SERIAL PRIMARY KEY,

\section{Introduction to Named Blocks}
Anonymous blocks (\texttt{DO \$\$}) evaluate procedural logic for single execution but are not persistently stored in the database schema. Persistent named blocks, referred to as subprograms, encapsulate procedural logic for reuse. In PostgreSQL, subprograms are classified into two primary constructs: \textbf{User-Defined Functions} and \textbf{Stored Procedures}.

\subsection{User-Defined Functions}
A named PL/pgSQL block that evaluates to a specific return type. Functions are invoked within expressions (e.g., a \texttt{SELECT} statement) and structurally require a \texttt{RETURN} or \texttt{RETURN QUERY} statement. They execute entirely within the transaction scope of the invoking query; they cannot execute Transaction Control Language (TCL) commands such as \texttt{COMMIT} or \texttt{ROLLBACK}.

\subsection{Stored Procedures}
A named PL/pgSQL block invoked via the \texttt{CALL} statement. Procedures do not evaluate to a direct return value and are capable of autonomous transaction control. This capability dictates their use for bulk data manipulation, schema migrations, and batch processing workflows.

% =======================================================
% SECTION: PARAMETER MODES
% =======================================================
\section{Parameter Modes}
Formal parameters in PL/pgSQL are assigned a mode that dictates the direction of data binding between the calling environment and the subprogram.

\begin{itemize}
    \item \textbf{\texttt{IN} Mode (Implicit Default):} Specifies that the formal parameter accepts an actual argument from the caller. It is lexically immutable (read-only) within the subprogram body.
    \item \textbf{\texttt{OUT} Mode:} Specifies an uninitialized variable bound to the caller's environment. The subprogram assigns a value to it, which is transmitted back upon termination.
    \item \textbf{\texttt{INOUT} Mode:} A hybrid parameter that accepts an initialized actual argument from the caller, permits state mutation within the subprogram body, and transmits the modified state back to the caller upon termination.
\end{itemize}

% =======================================================
% SECTION: DEFINING SUBPROGRAMS
% =======================================================
\section{Defining the Subprograms}
Before invoking them dynamically, subprograms must be compiled into the schema.

\begin{lstlisting}[language=SQL, caption={Scalar Function Example}]
CREATE OR REPLACE FUNCTION calculate_customer_lifetime_value(p_cust_id INT)
RETURNS NUMERIC AS $$
DECLARE
    v_total_spent NUMERIC;
BEGIN
    SELECT COALESCE(SUM(total_value), 0) INTO v_total_spent
    FROM orders
    WHERE customer_id = p_cust_id;
    
    RETURN v_total_spent;
END;
$$ LANGUAGE plpgsql;
\end{lstlisting}

\begin{lstlisting}[language=SQL, caption={Procedure Example with Mixed Modes and TCL}]
CREATE OR REPLACE PROCEDURE process_category_discount(
    p_category IN VARCHAR,      -- Read-only constraint
    p_discount_budget INOUT NUMERIC,   -- Mutated and returned to caller
    p_budget_status OUT VARCHAR -- Assigned and returned to caller
)
LANGUAGE plpgsql AS $$
DECLARE
    v_product_count INT;
BEGIN
    -- Utilize IN parameter for scalar subquery
    SELECT COUNT(*) INTO v_product_count
    FROM products
    WHERE category = p_category;
    
    -- Transform the INOUT parameter
    p_discount_budget := p_discount_budget - (v_product_count * 50.00);
    
    -- Initialize the OUT parameter
    IF p_discount_budget > 0 THEN
        p_budget_status := 'BUDGET_SURPLUS';
    ELSE
        p_budget_status := 'BUDGET_DEFICIT';
    END IF;
    
    -- Procedural Transaction Control
    COMMIT;
END;
$$;
\end{lstlisting}

% =======================================================
% SECTION: EXECUTION INSIDE ANONYMOUS BLOCKS
% =======================================================
\section{Execution Inside Anonymous Blocks}
This is the standard methodology for testing or dynamically invoking procedural logic without saving a new script to the database schema. Because anonymous blocks cannot return result sets directly to the client interface like a standard \texttt{SELECT} statement, you must utilize the \texttt{DECLARE} section to capture outputs and \texttt{RAISE NOTICE} to observe the results.

\subsection{Calling a Function}
Functions evaluate to a value. Inside a block, you must either assign this evaluated value to a declared variable using the \texttt{:=} operator, or use \texttt{SELECT ... INTO}.

\begin{lstlisting}[language=SQL, caption={Calling a Function}]
DO $$
DECLARE
    -- 1. Declare a variable to hold the function's return payload
    v_ltv NUMERIC;
BEGIN
    -- 2. Assign the function evaluation directly to the variable
    v_ltv := calculate_customer_lifetime_value(1);
    
    -- 3. Output the state
    RAISE NOTICE 'Customer LTV: %', v_ltv;
END;
$$;
\end{lstlisting}

\subsection{Calling a Procedure}
Procedures are invoked using the \texttt{CALL} statement. When a procedure utilizes \texttt{OUT} or \texttt{INOUT} parameters, you cannot pass raw literals into those parameter slots. You must pass \textbf{variables by reference} so the procedure has a memory address to write the modified state back to.

\begin{lstlisting}[language=SQL, caption={Calling a Procedure}]
DO $$
DECLARE
    -- 1. Initialize variables for INOUT parameters
    v_promo_budget NUMERIC := 10000.00; 
    
    -- 2. Declare uninitialized variables for OUT parameters
    v_status VARCHAR;  
BEGIN
    -- 3. Execute the procedure, passing the variables by reference
    CALL process_category_discount(
        'Electronics',   -- IN argument (literal is fine here)
        v_promo_budget,  -- INOUT argument (must be a variable)
        v_status         -- OUT argument (must be a variable)
    );
    
    -- 4. The variables in this block's scope now hold the mutated states
    RAISE NOTICE 'Remaining Budget: %, Status: %', v_promo_budget, v_status;
END;
$$;
\end{lstlisting}

% 1
    \item Write a \textbf{Stored Procedure} named \texttt{evaluate\_loyalty\_tiers} that accepts an \texttt{INOUT} parameter \texttt{p\_upgraded\_count} (initialized to 0). The procedure must loop through all customers, calculate their lifetime spending (sum of \texttt{total\_value} for \texttt{'PROCESSED'} orders), and upgrade their \texttt{loyalty\_tier} based on the following thresholds: $> \$10,000$ to \texttt{'Platinum'}, $> \$5,000$ to \texttt{'Gold'}, $> \$1,000$ to \texttt{'Silver'}. Do not downgrade anyone. If a customer's tier is changed, increment the \texttt{INOUT} counter.
    
    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE PROCEDURE evaluate_loyalty_tiers(INOUT p_upgraded_count INT) 
LANGUAGE plpgsql AS $$
DECLARE 
    v_cust RECORD;
BEGIN
    FOR v_cust IN 
        SELECT c.customer_id, c.loyalty_tier, COALESCE(SUM(o.total_value), 0) AS spend
        FROM customers c LEFT JOIN orders o ON c.customer_id = o.customer_id AND o.status = 'PROCESSED'
        GROUP BY c.customer_id, c.loyalty_tier 
    LOOP
        IF v_cust.spend > 10000 AND v_cust.loyalty_tier != 'Platinum' THEN
            UPDATE customers SET loyalty_tier = 'Platinum' WHERE customer_id = v_cust.customer_id;
            p_upgraded_count := p_upgraded_count + 1;
        ELSIF v_cust.spend > 5000 AND v_cust.loyalty_tier IN ('Bronze', 'Silver') THEN
            UPDATE customers SET loyalty_tier = 'Gold' WHERE customer_id = v_cust.customer_id;
            p_upgraded_count := p_upgraded_count + 1;
        ELSIF v_cust.spend > 1000 AND v_cust.loyalty_tier = 'Bronze' THEN
            UPDATE customers SET loyalty_tier = 'Silver' WHERE customer_id = v_cust.customer_id;
            p_upgraded_count := p_upgraded_count + 1;
        END IF;
    END LOOP;
    COMMIT;
END; 
$$;
    \end{lstlisting}
    \fi

% 2
    \item Create a \textbf{Stored Procedure} named \texttt{consolidate\_customer\_orders} that takes an \texttt{IN} parameter \texttt{p\_customer\_id} and an \texttt{OUT} parameter \texttt{p\_master\_order\_id}. Find all \texttt{'PENDING'} orders for this customer. If they have fewer than 2 pending orders, set the OUT parameter to \texttt{-1} and exit. Otherwise, designate the oldest pending order as the "master" order. Transfer all other pending \texttt{order\_items} to the master order, delete the emptied orders, and update the master order's \texttt{total\_value}.

    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE PROCEDURE consolidate_customer_orders(IN p_customer_id INT, OUT p_master_order_id INT) 
LANGUAGE plpgsql AS $$
DECLARE 
    v_ord RECORD;
BEGIN
    SELECT order_id INTO p_master_order_id FROM orders 
    WHERE customer_id = p_customer_id AND status = 'PENDING' ORDER BY order_date ASC LIMIT 1;

    IF NOT FOUND OR (SELECT COUNT(*) FROM orders WHERE customer_id = p_customer_id AND status = 'PENDING') < 2 THEN
        p_master_order_id := -1; RETURN;
    END IF;

    FOR v_ord IN SELECT order_id FROM orders WHERE customer_id = p_customer_id AND status = 'PENDING' AND order_id != p_master_order_id LOOP
        UPDATE order_items SET order_id = p_master_order_id WHERE order_id = v_ord.order_id;
        DELETE FROM orders WHERE order_id = v_ord.order_id;
    END LOOP;

    UPDATE orders SET total_value = (SELECT COALESCE(SUM(quantity * unit_price), 0) FROM order_items WHERE order_id = p_master_order_id) 
    WHERE order_id = p_master_order_id;
    COMMIT;
END; 
$$;
    \end{lstlisting}
    \fi

% 3
    \item Write a \textbf{Stored Procedure} named \texttt{dynamic\_pricing\_adjustment} with two \texttt{OUT} parameters: \texttt{p\_increased\_count} and \texttt{p\_decreased\_count}. Iterate through all products. If the total quantity ever sold is greater than 100, increase the \texttt{base\_price} by 10\%. If the total sold is strictly less than 10, decrease the \texttt{base\_price} by 15\%. Track the number of products modified.
    
    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE PROCEDURE dynamic_pricing_adjustment(OUT p_increased_count INT, OUT p_decreased_count INT) 
LANGUAGE plpgsql AS $$
DECLARE 
    v_prod RECORD;
BEGIN
    p_increased_count := 0; p_decreased_count := 0;

    FOR v_prod IN SELECT p.product_id, COALESCE(SUM(oi.quantity), 0) AS qty 
                  FROM products p LEFT JOIN order_items oi USING (product_id) GROUP BY p.product_id LOOP
        IF v_prod.qty > 100 THEN
            UPDATE products SET base_price = base_price * 1.10 WHERE product_id = v_prod.product_id;
            p_increased_count := p_increased_count + 1;
        ELSIF v_prod.qty < 10 THEN
            UPDATE products SET base_price = base_price * 0.85 WHERE product_id = v_prod.product_id;
            p_decreased_count := p_decreased_count + 1;
        END IF;
    END LOOP;
    COMMIT;
END; 
$$;
    \end{lstlisting}
    \fi

% 4
    \item Design a \textbf{Stored Procedure} \texttt{cancel\_order\_and\_restore(IN p\_order\_id, OUT p\_refund\_amount)}. Verify the order is not already \texttt{'CANCELLED'}. Loop through every item in that order. Add the purchased quantity back to \texttt{products.stock\_quantity}, insert a record into \texttt{restock\_logs}, and accumulate the financial value for the refund. Finally, update the order status to \texttt{'CANCELLED'}.

    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE PROCEDURE cancel_order_and_restore(IN p_order_id INT, OUT p_refund_amount NUMERIC) 
LANGUAGE plpgsql AS $$
DECLARE
    v_item RECORD;
BEGIN
    p_refund_amount := 0.00;

    IF (SELECT status FROM orders WHERE order_id = p_order_id) = 'CANCELLED' THEN
        RAISE EXCEPTION 'Order % is already cancelled.', p_order_id;
    END IF;

    FOR v_item IN SELECT product_id, quantity, (quantity * unit_price) AS cost FROM order_items WHERE order_id = p_order_id LOOP
        UPDATE products SET stock_quantity = stock_quantity + v_item.quantity WHERE product_id = v_item.product_id;
        INSERT INTO restock_logs (product_id, restock_date, added_quantity) VALUES (v_item.product_id, CURRENT_DATE, v_item.quantity);
        p_refund_amount := p_refund_amount + v_item.cost;
    END LOOP;

    UPDATE orders SET status = 'CANCELLED' WHERE order_id = p_order_id;
    COMMIT;
END; 
$$;
    \end{lstlisting}
    \fi

% 5
    \item Write a \textbf{Table-Valued Function} \texttt{get\_cross\_sell\_recommendations(p\_target\_product\_id)}. Identify all orders containing this target product. Find all \textit{other} products purchased within those exact same orders. Return a table of \texttt{(rec\_product\_id, rec\_name, co\_buy\_count)} for the top 3 most frequently co-purchased items, sorted descending by count.

    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE FUNCTION get_cross_sell_recommendations(p_target_product_id INT)
RETURNS TABLE (rec_product_id INT, rec_name VARCHAR, co_buy_count INT) 
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT oi.product_id, p.name, COUNT(*)::INT as frequency
    FROM order_items oi JOIN products p USING (product_id)
    WHERE oi.order_id IN (SELECT order_id FROM order_items WHERE product_id = p_target_product_id)
      AND oi.product_id != p_target_product_id
    GROUP BY oi.product_id, p.name ORDER BY frequency DESC LIMIT 3;
END; 
$$;
    \end{lstlisting}
    \fi

% 6
    \item Implement a \textbf{Table-Valued Function} \texttt{category\_performance\_report} with no parameters. Compute the total gross revenue (from \texttt{'PROCESSED'} orders) and identify the single best-selling product (by quantity) for every category. Return \texttt{(category\_name, total\_gross\_revenue, best\_selling\_product\_name)}.

    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE FUNCTION category_performance_report()
RETURNS TABLE (category_name VARCHAR, total_gross_revenue NUMERIC, best_selling_product_name VARCHAR) 
LANGUAGE plpgsql AS $$
BEGIN
    -- Optimized: Return query directly using CTEs, completely avoiding procedural loops
    RETURN QUERY 
    WITH CatRev AS (
        SELECT p.category, COALESCE(SUM(oi.quantity * oi.unit_price), 0) AS rev
        FROM products p 
        LEFT JOIN order_items oi USING (product_id) 
        LEFT JOIN orders o ON oi.order_id = o.order_id AND o.status = 'PROCESSED'
        GROUP BY p.category
    ),
    BestSellers AS (
        SELECT DISTINCT ON (p.category) p.category, p.name 
        FROM products p JOIN order_items oi USING (product_id) JOIN orders o USING (order_id)
        WHERE o.status = 'PROCESSED'
        GROUP BY p.category, p.name ORDER BY p.category, SUM(oi.quantity) DESC
    )
    SELECT cr.category, cr.rev, COALESCE(bs.name, 'N/A'::VARCHAR)
    FROM CatRev cr LEFT JOIN BestSellers bs USING (category);
END; 
$$;

    \item Write a \textbf{Stored Procedure} \texttt{auto\_restock\_depleted(IN p\_threshold, IN p\_restock\_amount, OUT p\_total\_cost)}. Loop through products where \texttt{stock\_quantity} $<$ \texttt{p\_threshold}. Increase the stock by \texttt{p\_restock\_amount}, insert a log into \texttt{restock\_logs}, and accumulate the total restocking cost (wholesale cost is 40\% of \texttt{base\_price}).

    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE PROCEDURE auto_restock_depleted(IN p_threshold INT, IN p_restock_amount INT, OUT p_total_cost NUMERIC) 
LANGUAGE plpgsql AS $$
DECLARE
    v_prod RECORD;
BEGIN
    p_total_cost := 0.00;

    FOR v_prod IN SELECT product_id, base_price FROM products WHERE stock_quantity < p_threshold LOOP
        UPDATE products SET stock_quantity = stock_quantity + p_restock_amount WHERE product_id = v_prod.product_id;
        INSERT INTO restock_logs (product_id, restock_date, added_quantity) VALUES (v_prod.product_id, CURRENT_DATE, p_restock_amount);
        p_total_cost := p_total_cost + ((v_prod.base_price * 0.40) * p_restock_amount);
    END LOOP;
    COMMIT;
END; 
$$;
    \end{lstlisting}

    \item Develop a \textbf{Stored Procedure} \texttt{reward\_vip\_customers(IN p\_top\_n)}. Identify the top N customers by total lifetime spending (\texttt{'PROCESSED'} orders). For each VIP, find their most expensive purchased product and generate a complimentary order containing 1 unit of that product at a \texttt{unit\_price} of 0. Deduct the stock.

    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE PROCEDURE reward_vip_customers(IN p_top_n INT) 
LANGUAGE plpgsql AS $$
DECLARE
    v_cust RECORD; v_new_order_id INT;
BEGIN
    FOR v_cust IN 
        SELECT c.customer_id, 
               (SELECT oi.product_id FROM order_items oi JOIN orders o USING (order_id) 
                WHERE o.customer_id = c.customer_id ORDER BY oi.unit_price DESC LIMIT 1) AS fav_prod
        FROM orders c WHERE status = 'PROCESSED' GROUP BY c.customer_id ORDER BY SUM(total_value) DESC LIMIT p_top_n
    LOOP
        IF v_cust.fav_prod IS NOT NULL THEN
            INSERT INTO orders (customer_id, order_date, status, total_value) 
            VALUES (v_cust.customer_id, CURRENT_DATE, 'PROCESSED', 0.00) RETURNING order_id INTO v_new_order_id;
            
            INSERT INTO order_items (order_id, product_id, quantity, unit_price) 
            VALUES (v_new_order_id, v_cust.fav_prod, 1, 0.00);
            
            UPDATE products SET stock_quantity = stock_quantity - 1 WHERE product_id = v_cust.fav_prod;
        END IF;
    END LOOP;
    COMMIT;
END; 
$$;
    \end{lstlisting}
    \fi

    \item Create a \textbf{Scalar Function} \texttt{project\_future\_revenue(IN p\_months\_ahead)}. Calculate average daily revenue over the last 30 days and compare it to the prior 30 days to get a growth rate. Using the compound interest formula, project the total revenue for the 30-day period occurring exactly \texttt{p\_months\_ahead} in the future.

    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE FUNCTION project_future_revenue(p_months_ahead INT) RETURNS NUMERIC 
LANGUAGE plpgsql AS $$
DECLARE
    v_recent NUMERIC; v_past NUMERIC; v_rate NUMERIC;
BEGIN
    SELECT COALESCE(SUM(total_value), 0) INTO v_recent FROM orders WHERE order_date >= CURRENT_DATE - 30;
    SELECT COALESCE(SUM(total_value), 0) INTO v_past FROM orders WHERE order_date >= CURRENT_DATE - 60 AND order_date < CURRENT_DATE - 30;

    v_rate := CASE WHEN v_past = 0 THEN 0 ELSE (v_recent - v_past) / v_past END;
    RETURN ROUND(v_recent * POWER((1.0 + v_rate), p_months_ahead), 2);
END; 
$$;
    \end{lstlisting}
    \fi

    \item Implement a \textbf{Stored Procedure} \texttt{apply\_bulk\_discounts(IN p\_min\_qty, IN p\_discount\_pct)}. Iterate through all \texttt{'PENDING'} orders. If an item's requested \texttt{quantity} $\geq$ \texttt{p\_min\_qty}, calculate the discount and update its \texttt{unit\_price}. Recalculate and update the order's \texttt{total\_value} if modified.

    \ifshowsolutions
    \textbf{Solution:}
    \begin{lstlisting}[language=SQL]
CREATE OR REPLACE PROCEDURE apply_bulk_discounts(IN p_min_qty INT, IN p_discount_pct NUMERIC) 
LANGUAGE plpgsql AS $$
DECLARE
    v_ord RECORD; v_modified BOOLEAN;
BEGIN
    FOR v_ord IN SELECT order_id FROM orders WHERE status = 'PENDING' LOOP
        v_modified := FALSE;
        
        -- Update items and track if changes were made
        UPDATE order_items SET unit_price = unit_price * (1.0 - (p_discount_pct / 100.0))
        WHERE order_id = v_ord.order_id AND quantity >= p_min_qty RETURNING order_id INTO v_modified;
        
        -- Only recalculate totals for orders that were actually modified
        IF v_modified THEN
            UPDATE orders SET total_value = (
                SELECT COALESCE(SUM(quantity * unit_price), 0) FROM order_items WHERE order_id = v_ord.order_id
            ) WHERE order_id = v_ord.order_id;
        END IF;
    END LOOP;
    COMMIT;
END; 
$$;
    \end{lstlisting}
    \fi

    \item Design a \textbf{Stored Procedure} \texttt{anonymize\_inactive\_customers(OUT p\_anonymized\_count)}. If a customer has never placed an order or their last order was over 730 days ago, execute an anonymization process: change their \texttt{name} to \texttt{'ANONYMIZED'}, \texttt{region} to \texttt{'UNKNOWN'}, and \texttt{loyalty\_tier} to \texttt{'Bronze'}. Count updated rows.

\ifshowsolutions
\textbf{Solution:}
\begin{lstlisting}[language=SQL]
CREATE OR REPLACE PROCEDURE anonymize_inactive_customers(OUT p_anonymized_count INT) 
LANGUAGE plpgsql AS $$
DECLARE
    v_cust RECORD;
BEGIN
    p_anonymized_count := 0;

    FOR v_cust IN 
        SELECT c.customer_id, COALESCE(CURRENT_DATE - MAX(o.order_date), 9999) AS days_inactive
        FROM customers c LEFT JOIN orders o USING (customer_id) GROUP BY c.customer_id
    LOOP
        IF v_cust.days_inactive > 730 THEN
            UPDATE customers SET name = 'ANONYMIZED', region = 'UNKNOWN', loyalty_tier = 'Bronze'
            WHERE customer_id = v_cust.customer_id;
            p_anonymized_count := p_anonymized_count + 1;
        END IF;
    END LOOP;
    COMMIT;
END; 
$$;
\end{lstlisting}
\fi
